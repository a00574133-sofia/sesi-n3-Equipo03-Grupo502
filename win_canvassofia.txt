import tkinter as tk
from tkinter import ttk, messagebox
from datetime import datetime

import requests
from matplotlib.figure import Figure
from matplotlib.backends.backend_tkagg import FigureCanvasTkAgg


def fetch_data():
    """
    Obtiene de Open-Meteo las temperaturas horarias de León, Gto
    de las últimas 24 horas. Devuelve (horas, temperaturas).
    """
    try:
        url = (
            "https://api.open-meteo.com/v1/forecast"
            "?latitude=21.12&longitude=-101.68"
            "&hourly=temperature_2m&past_days=1&forecast_days=1"
            "&timezone=America%2FMexico_City"
        )
        response = requests.get(url, timeout=15)
        response.raise_for_status()
        data = response.json()

        temps = data["hourly"]["temperature_2m"]

# ---------- VERSIÓN 4: ÁREA + BARRAS HORIZONTALES ----------
VERDE = "#2a9d8f"
NARANJA = "#e76f51"


def create_line_chart(horas, temps):
    """Gráfica de área (línea con relleno)."""
    fig = Figure(figsize=(6, 3), dpi=100)
    ax = fig.add_subplot(111)
    x = list(range(len(horas)))
    ax.fill_between(x, temps, min(temps) - 1, color=VERDE, alpha=0.35)
    ax.plot(x, temps, color=VERDE, linewidth=2.5)
    ax.set_ylim(min(temps) - 1, max(temps) + 1)
    ax.set_xlim(0, len(horas) - 1)
    ax.set_xticks(range(0, len(horas), 3))
    ax.set_xticklabels(horas[::3])
    ax.set_title("Temperatura en León (área)", fontsize=11, fontweight="bold")
    ax.set_xlabel("Hora")
    ax.set_ylabel("°C")
    ax.tick_params(axis="x", rotation=45, labelsize=7)
    ax.grid(alpha=0.3)
    fig.tight_layout()
    return fig


def create_bar_chart(horas, temps):
    """Barras horizontales: una fila por hora, la más reciente arriba."""
    fig = Figure(figsize=(6, 4.2), dpi=100)
    ax = fig.add_subplot(111)
    ax.barh(range(len(horas)), temps, color=NARANJA, height=0.75)
    ax.set_yticks(range(len(horas)))
    ax.set_yticklabels(horas, fontsize=6)
    ax.invert_yaxis()
    ax.set_title("Temperatura en León (barras horizontales)", fontsize=11,
                 fontweight="bold")
    ax.set_xlabel("°C")
    ax.set_ylabel("Hora")
    ax.tick_params(axis="x", labelsize=7)
    ax.grid(axis="x", alpha=0.3)
    ax.set_axisbelow(True)
    fig.tight_layout()
    return fig


def mostrar_graficas(frm, horas, temps):
    """Inserta las dos gráficas en el frame."""
    for fig in (create_line_chart(horas, temps), create_bar_chart(horas, temps)):
        canvas = FigureCanvasTkAgg(fig, master=frm)
        canvas.draw()
        canvas.get_tk_widget().pack(pady=10, fill="x")


def open_win_canvas(parent: tk.Tk):
    """Crea la ventana secundaria con gráficas de la API."""
    win = tk.Toplevel(parent)
    win.title("Canvas con API (Open-Meteo) y gráficas")
    win.geometry("960x900")

    top = ttk.Frame(win, padding=12)
    top.pack(fill="x")

    charts_frame = ttk.Frame(win, padding=12)
    charts_frame.pack(fill="both", expand=True)

    def cargar():
        horas, temps = fetch_data()
        if horas and temps:
            for w in charts_frame.winfo_children():
                w.destroy()
            mostrar_graficas(charts_frame, horas, temps)

    ttk.Button(top, text="Cargar y mostrar gráficas", command=cargar).pack()


if __name__ == "__main__":
    root = tk.Tk()
    root.title("Prueba win_canvas")
    ttk.Button(
        root, text="Abrir ventana Canvas",
        command=lambda: open_win_canvas(root),
    ).pack(pady=20)
    root.mainloop()
